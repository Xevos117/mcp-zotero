+// Shared helpers of the agent smoke hooks (settings: .claude/agent-smoke/settings.json). Self-contained on
+// purpose: on pull requests claude-code-action restores .claude/ from the base branch, so these files are the
+// trusted base version while everything else in the checkout comes from the PR.
+// Evidence lives in <project>/agent-audit/, outside agent-out/, where the agent cannot write (its Edit/Write
+// rules and both allowed commands are confined to agent-out/).
+import { appendFileSync, existsSync, mkdirSync, readFileSync } from "node:fs";
+import { join } from "node:path";
+
+export const ZOTERO_KEY = /^[23456789ABCDEFGHIJKLMNPQRSTUVWXYZ]{8}$/;
+/** Valid-shape keys that exist in no library (F10, T-delete): deleting them is harmless. */
+export const FAKE_KEYS = new Set(["ZZZZZZZZ", "YYYYYYYY"]);
+
+export function auditDir(input) {
+  return join(process.env.CLAUDE_PROJECT_DIR || input?.cwd || process.cwd(), "agent-audit");
+}
+
+export function readStdin() {
+  try {
+    return JSON.parse(readFileSync(0, "utf8") || "{}");
+  } catch {
+    return {};
+  }
+}
+
+export function append(input, file, entry) {
+  const dir = auditDir(input);
+  mkdirSync(dir, { recursive: true });
+  appendFileSync(join(dir, file), JSON.stringify({ ts: new Date().toISOString(), ...entry }) + "\n");
+}
+
+export function readJsonl(path) {
+  if (!existsSync(path)) return [];
+  return readFileSync(path, "utf8")
+    .split("\n")
+    .filter((l) => l.trim())
+    .map((l) => {
+      try {
+        return JSON.parse(l);
+      } catch {
+        return null;
+      }
+    })
+    .filter(Boolean);
+}
+
+/** PreToolUse denial with a reason the agent sees; also logged in agent-audit/denials.jsonl. */
+export function deny(input, reason) {
+  append(input, "denials.jsonl", { tool_name: input?.tool_name, tool_use_id: input?.tool_use_id, tool_input: input?.tool_input, reason });
+  process.stdout.write(JSON.stringify({ hookSpecificOutput: { hookEventName: "PreToolUse", permissionDecision: "deny", permissionDecisionReason: reason } }));
+  process.exit(0);
+}
+
+/** Text of a tool response as the hooks receive it (string, MCP content blocks, {content}, Bash {stdout, stderr}). */
+export function responseText(response) {
+  if (response == null) return "";
+  if (typeof response === "string") return response;
+  if (Array.isArray(response)) return response.map((b) => (typeof b === "string" ? b : b?.type === "text" ? b.text : "")).join("");
+  if (typeof response === "object") {
+    if (Array.isArray(response.content)) return responseText(response.content);
+    if ("stdout" in response || "stderr" in response) return [response.stdout, response.stderr].filter(Boolean).join("\n");
+  }
+  return JSON.stringify(response);
+}
+
+/** Keys created by successful Zotero calls recorded in the audit log (structured result fields only). */
+export function createdKeys(entries) {
+  const created = new Set();
+  const add = (k) => typeof k === "string" && ZOTERO_KEY.test(k) && created.add(k);
+  for (const e of entries) {
+    if (e.is_error || !String(e.tool_name ?? "").startsWith("mcp__zotero__")) continue;
+    let json;
+    try {
+      json = JSON.parse(e.text);
+    } catch {
+      continue;
+    }
+    if (!json || typeof json !== "object" || json.error) continue;
+    switch (e.tool_name.slice("mcp__zotero__".length)) {
+      case "create_collection":
+        add(json.collection_key);
+        break;
+      case "add_items":
+      case "add_items_by_doi":
+        for (const x of json.success ?? []) add(x?.item_key);
+        for (const x of json.pdf_results ?? []) add(x?.attachment_key);
+        break;
+      case "add_linked_url_attachment":
+      case "import_pdf_to_zotero":
+        add(json.item_key);
+        break;
+      case "find_and_attach_pdfs":
+        for (const x of json.results ?? []) add(x?.attachment_key);
+        break;
+    }
+  }
+  return created;
+}
+
+// --- Bash allowlist (bash-guard.mjs) ---
+
+const FILE = (ext) => `agent-out/[A-Za-z0-9._-]+\\.${ext}`;
+// single-quoted JSON spec: no quote, backslash, newline or shell metacharacter inside either
+const SPEC = "'[^'\\\\\\n\\r$`;&|<>]*'";
+const STYLE = "(?:apa|ieee|vancouver|harvard|chicago)";
+const LIBRARY = "(?:(?:users|groups)/[0-9]+|[0-9]+)";
+const ALLOWED = [
+  new RegExp(`^node test/live/agent/make-docx\\.mjs(?: ${FILE("docx")})+ ${SPEC}$`),
+  new RegExp(`^node test/live/agent/inject\\.mjs ${FILE("docx")} ${FILE("docx")} ${FILE("json")} ${LIBRARY}(?: ${STYLE})?$`),
+];
+export const BASH_USAGE =
+  "Only these Bash commands are allowed, exactly in this form, one per call: " +
+  "node test/live/agent/make-docx.mjs agent-out/<name>.docx [agent-out/<name>.docx ...] '<json spec>' | " +
+  "node test/live/agent/inject.mjs agent-out/<in>.docx agent-out/<out>.docx agent-out/<metadata>.json <users/ID|groups/ID> [style]. " +
+  "No other programs, options, variables, quotes, chaining, pipes or redirections.";
+
+export function bashDecision(toolInput) {
+  const command = toolInput?.command;
+  if (typeof command !== "string") return "no command";
+  if (toolInput.run_in_background) return "background commands are not allowed";
+  if (!ALLOWED.some((re) => re.test(command))) return "command not allowed";
+  return null;
+}
+
+// --- Delete policy (delete-guard.mjs) ---
+
+export function deleteDecision(toolName, toolInput, context, created) {
+  if (!/^mcp__zotero__delete_(items|collection)$/.test(String(toolName))) return "not a delete tool";
+  if (!context) return "run context missing (agent-audit/context.json): deletions are not allowed";
+  const tool = String(toolName).replace(/^mcp__zotero__/, "");
+  if (tool === "delete_collection" && context.safe_mode) return "delete_collection is never allowed in safe mode";
+  if (toolInput?.library_type === "group" && (context.safe_mode || String(toolInput.library_id) !== String(context.group_id))) return "deletions only in the test library";
+  const keys = tool === "delete_items" ? toolInput?.item_keys : [toolInput?.collection_key];
+  if (!Array.isArray(keys)) return null; // not an array: the server rejects it
+  for (const k of keys) {
+    if (typeof k !== "string" || !ZOTERO_KEY.test(k)) return "malformed key: never pass one to a delete tool";
+    if (!created.has(k) && !FAKE_KEYS.has(k)) return `${k} was not created by this run`;
+  }
+  return null;
+}

+## Step 4 — Exploratory scenarios (max 5)
+
+Design **up to 5** extra scenarios `X1`…`X5` that probe the robustness of **what this change touches**:
+read `agent-out/diff.txt`, pick the tools, arguments or code paths it modifies, and try the inputs a
+careless caller would send: `null`, missing or wrong-type values, empty strings and arrays,
+whitespace, very long strings, unicode, boundary numbers (0, negative, huge, fractional), duplicates,
+valid and invalid values mixed in one call. Example: the diff changes the input of
+`add_items_by_doi` → send `dois: ["", "  ", null]`, a 2000-character DOI, a DOI with unicode, a valid
+and an invalid DOI together. If the diff touches no tool (docs, CI only), pick the tools closest to it
+or run none and say so in the report.
+
+- Same rules as everything else: only data this run creates (prefix and run tag on every title you
+  write), respect `safe_mode` and `unsafe_operations`, never more than 5 scenarios.
+- Never pass a malformed or foreign key to a **write** tool (`add_*`, `import_pdf_to_zotero`,
+  `find_and_attach_pdfs` without `dry_run`, `create_collection`, `delete_*`): malformed keys made
+  Zotero ignore the filter and return the whole library in 2.0.0. Probe key validation with read tools
+  (`get_items_details`, `get_item_fulltext`, `get_collection_items`) or with `ZZZZZZZZ` / `YYYYYYYY`.
+- Expected behaviour: a readable result or a readable error with `isError`; never a crash, a closed
+  connection, a stack trace, `[object Object]`, or a raw `undefined`/`NaN` in the text.
+- Verdict per scenario: `ok` (behaves as expected), `bug` (crash, data touched that should not be,
+  silently wrong result, unreadable error), `unclear` (debatable behaviour worth a human look). **A
+  `bug` fails the check**; `unclear` is shown as a warning.
+- Record the calls exactly as you sent them (`tool` and `args`): the verifier looks each one up in the
+  audit log. For inject.js use `{"tool": "Bash", "args": {"command": "<the exact command>"}}`.
+
+The verifier also scans **every** Zotero tool result of the run, fixed path included, for the
+failures listed above, so a crash anywhere fails the check whatever you claim.

+# CLAUDE.md — instructions for agents working on mcp-zotero
+
+MCP server for the Zotero Web API (TypeScript, ESM, Node >= 22), published on npm as
+`@xevos117/mcp-zotero`. Source in `src/`, compiled to `build/` with `tsc` (TypeScript 7).
+The product skill lives in `skills/zotero-skill-mcp-integrations/` (standalone `inject.js`,
+shipped as a release zip, not in the npm package).
+
+## Hard rules
+
+- **Timeouts everywhere.** Every command that can hang runs under `timeout N` (git network ops,
+  npm, vitest, node scripts, gh). Every CI job and step has `timeout-minutes`. Every test has a
+  timeout; every network call in code uses `AbortSignal.timeout`. A bound that is hit is a failure
+  to investigate, not something to retry blindly.
+- **Never merge, publish or release.** No `gh pr merge`, `npm publish`, `gh release create`, tag
+  pushes or version bumps unless the owner asks for that specific action. Do not change repository
+  settings (rulesets, environments, secrets, workflow permissions).
+- **Never print secrets.** Zotero / Anthropic / npm keys are read only from the environment or from
+  files passed inline (`VAR="$(cat file)" cmd`); never echo them, log them, commit them or paste
+  them in PRs, issues or reports. Scrub keys, user ids and emails from anything you publish.
+- **Real Zotero libraries are someone's data.** Live runs only create items/collections prefixed
+  `mzt-` and tagged `mzt-run:<runId>`, and delete only what that run created.
+
+## Conventions
+
+- Commit messages, PR titles/descriptions and `test/TEST_CASES.md` are written in **Italian**.
+  Code comments and identifiers are in English. Keep diffs focused; match the surrounding style.
+- Never edit `build/` by hand; it is gitignored and produced by `npm run build`.
+- Tool errors go through `formatErrorResponse` (`isError: true`); empty results through
+  `formatEmptyResult` (no `isError`). Deletions are gated by `UNSAFE_OPERATIONS` (`none|items|all`).
+- `.claude/` is ignored except `.claude/skills/` (versioned project skills). `.npmignore` keeps
+  `.claude/` and `CLAUDE.md` out of the npm package; `package.json` `files` is `["build"]`.
